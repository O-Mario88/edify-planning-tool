# Cache client for django.core.cache.backends.redis (config/settings/base.py).
#
# This was missing, and its absence was invisible: the cache probe wraps the
# import and the ping in one `except Exception`, so a ModuleNotFoundError was
# reported as "Redis unreachable at <url>" — indistinguishable from a network
# failure. Production ran on a per-process LocMemCache the entire time while
# appearing to have merely mis-set REDIS_URL.
#
# Without it the cache silently degrades: the login throttle counts per worker
# rather than globally, and nothing is shared between the web instances or with
# the scheduler. That makes it a correctness dependency for running more than
# one instance, not a performance one.
redis==7.1.0
